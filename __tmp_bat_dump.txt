@echo off
title GhostWriter All-in-One Launcher
setlocal

:: 設定編碼為 UTF-8 以支援中文顯示
chcp 65001 > nul

echo ===================================================
echo   GhostWriter 👻 - 手機輸入橋接 + 圓盤選單工具包
echo ===================================================
echo.

:: 檢查虛擬環境是否存在
if not exist ".venv" (
    echo [錯誤] 找不到 .venv 虛擬環境！
    echo 正在嘗試建立並安裝依賴...
    python -m venv .venv
    if exist ".venv\Scripts\activate.bat" (
        call ".venv\Scripts\activate.bat"
    ) else (
        echo [錯誤] 找不到 .venv\Scripts\activate.bat
        pause
        exit /b 1
    )
    python -m pip install -r ghostwriter/requirements.txt
) else (
    echo [系統] 正在啟動虛擬環境...
    if exist ".venv\Scripts\activate.bat" (
        call ".venv\Scripts\activate.bat"
    ) else (
        echo [錯誤] 找不到 .venv\Scripts\activate.bat
        pause
        exit /b 1
    )
)

echo [系統] 正在啟動 GhostWriter 整合服務...
echo [提示] 啟動後：
echo        1. 手機掃描 QR Code 即可打字
echo        2. 電腦按下 Ctrl + 滑鼠中鍵 呼叫圓盤選單
echo.

:: 執行啟動器
python launcher.py

if %errorlevel% neq 0 (
    echo.
    echo [警告] 程式異常退出 (Exit Code: %errorlevel%)
    pause
)

endlocal
